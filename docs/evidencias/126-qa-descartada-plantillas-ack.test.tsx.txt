const saveInvitationTemplateReal = jest.requireActual<typeof import('@/app/actions/invitacion-digital-templates')>('@/app/actions/invitacion-digital-templates').saveInvitationTemplate;
mockSaveInvitationTemplate = jest.fn((...args: unknown[]) => saveInvitationTemplateReal(...args as Parameters<typeof saveInvitationTemplateReal>));

const storeKey = 'invitacion-digital-templates.json';

beforeAll(() => { Object.defineProperty(globalThis, 'IS_REACT_ACT_ENVIRONMENT', { configurable: true, value: true }); });

async function renderScreen() {
  const host = document.body.appendChild(document.createElement('div'));
  const root = createRoot(host);
  await act(async () => { root.render(<InvitationTemplatesPage />); });
  return { host, root };
}

async function createClick(host: HTMLElement) {
  const button = Array.from(host.querySelectorAll('button')).find((item) => item.textContent?.includes('Crear Nueva Plantilla')) as HTMLButtonElement;
  if (!button) throw new Error('No se encontró el botón visible para crear plantilla.');
  await act(async () => { button.click(); });
}

test('respuesta perdida tras persistir: reintento del mismo flujo no duplica plantilla', async () => {
  mockSaveInvitationTemplate.mockImplementationOnce(async (payload) => {
      await saveInvitationTemplateReal(payload);
      throw new Error('respuesta perdida simulada');
    });
  const { host, root } = await renderScreen();

  await createClick(host);
  const firstStore = mockMemory.get(storeKey);
  expect(mockSaveInvitationTemplate).toHaveBeenCalledTimes(1);
  expect(mockWriteData).toHaveBeenCalled();
  expect(firstStore).toHaveLength(1);
  expect(mockToast).toHaveBeenCalledWith(expect.objectContaining({ title: 'Error', description: expect.stringContaining('respuesta perdida simulada'), variant: 'destructive' }));

  await createClick(host);
  const retryStore = mockMemory.get(storeKey);
  expect(retryStore).toHaveLength(1);
  expect(retryStore[0].id).toBe(firstStore[0].id);
  expect(retryStore[0]).toEqual(firstStore[0]);
  expect(mockSaveInvitationTemplate).toHaveBeenCalledTimes(2);
  expect(mockSaveInvitationTemplate.mock.calls[1][0]).toEqual(mockSaveInvitationTemplate.mock.calls[0][0]);

  act(() => root.unmount()); host.remove();
});

test('dos creaciones deliberadas en éxito ordinario generan dos plantillas', async () => {
  const { host, root } = await renderScreen();
  await createClick(host);
  await createClick(host);

  const saved = mockMemory.get(storeKey);
  expect(saved).toHaveLength(2);
  expect(saved[0].id).not.toBe(saved[1].id);
  expect(saved.map((item: any) => item.name)).toEqual(['Nueva Plantilla 1', 'Nueva Plantilla 2']);
  act(() => root.unmount()); host.remove();
});
